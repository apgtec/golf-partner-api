    Operations¶

    Ready-to-use operation documents, one per file under operations/. Each is validated against the schema on every change. Fill in the variables shown in the Integration guide.

    UpsertTournament¶

    # Tournament structure and status. Idempotent upsert keyed by your externalId.
# Returns the Bolt6 tournament id you use for upsertCourse and upsertRounds.
mutation Query($input: TournamentInput!) {
    upsertTournament(input: $input) {
        accepted
        message
        ids { kind externalId id }
        rejected { kind externalId message }
    }
}

    UpsertCourse¶

    # Course, holes, par and hole length (metres), and the UTM zone the course sits
# in. Needs the Bolt6 tournamentId from upsertTournament; returns the course id
# you use in upsertRounds. Par and length are yours; Bolt6 stores them verbatim.
mutation Query($input: CourseInput!) {
    upsertCourse(input: $input) {
        accepted
        message
        ids { kind externalId id }
        rejected { kind externalId message }
    }
}

    UpsertRounds¶

    # Rounds of a tournament, each on a course. Needs the Bolt6 tournamentId and
# courseIds from the previous steps; returns the round ids every read and every
# stroke is keyed by. Rounds not listed are left untouched.
mutation Query($input: RoundsInput!) {
    upsertRounds(input: $input) {
        accepted
        message
        ids { kind externalId id }
        rejected { kind externalId message }
    }
}

    UpsertGroups¶

    # Groups (pairings) and their players for a round. Needs the Bolt6 roundId;
# returns group and player ids for upsertStroke. Groups not listed are left
# untouched.
mutation Query($input: GroupsInput!) {
    upsertGroups(input: $input) {
        accepted
        message
        ids { kind externalId id }
        rejected { kind externalId message }
    }
}

    UpsertStroke¶

    # A stroke a player is about to play, or has just played. Needs the Bolt6
# roundId, groupId and playerId; carries your externalId for the stroke and
# returns its Bolt6 id (needed for retractStroke).
#
# This reports stroke STATE, not ball position: fromSurface is the surface the
# ball lies on. Send one per stroke at HIT (the default), or one per status
# change if you have pre-shot events; resending is safe.
mutation Query($input: StrokeInput!) {
    upsertStroke(input: $input) {
        accepted
        message
        ids { kind externalId id }
        rejected { kind externalId message }
    }
}

    RetractStroke¶

    # Withdraw a stroke reported in error, by its Bolt6 strokeId. Bolt6 marks its
# positions retracted and publishes that as a new revision.
mutation Query($input: RetractStrokeInput!) {
    retractStroke(input: $input) {
        accepted
        message
    }
}

    GetCourse¶

    # The course and, critically, the CRS declaration for every coordinate on this
# contract. Fetch once per round before consuming positions.
query Query(
    $roundId: ID!
) {
    course(roundId: $roundId) {
        id
        externalId
        name
        utmZone
        utmHemisphere
        epsg
        verticalEpsg
        par
        frontPar
        backPar
        length
        holes {
            number
            par
            length
        }
    }
}

    SubscribeToBallPositionEvents¶

    # Resumable ball-position stream. Queue semantics.
#
# Pass afterRevision: 0 to replay a round from the beginning; otherwise pass
# the highest revision you have durably processed. Revisions increase across
# both inserts and updates, so corrections and retractions arrive here too -
# check `retracted` on every delivery.
subscription Query(
    $roundId: ID!
    $afterRevision: Int!
    $batchSize: Int
) {
    ballPositionEvents(roundId: $roundId, afterRevision: $afterRevision, batchSize: $batchSize) {
        revision
        strokeId
        strokeExternalId
        strokeNum
        strokeKind
        playerId
        playerExternalId
        groupId
        groupExternalId
        hole
        holeOrder
        easting
        northing
        elevation
        surface
        subSurface
        state
        inTheHole
        retracted
        observedAt
        recordedAt
    }
}
